import{spawn,execFileSync}from'node:child_process';import{openSync,closeSync,readFileSync,writeFileSync,unlinkSync,existsSync}from'node:fs';
const[label,heapText,script,...args]=process.argv.slice(2),heap=Number(heapText),root='/private/tmp/actor-week-optimization-20261029',lockPath='/private/tmp/actor-cinema-normal-cohort.lock';if(!label||!script||heap<512||heap>2048)throw Error('Bounded job arguments required');
let lock;try{lock=openSync(lockPath,'wx');writeFileSync(lock,String(process.pid))}catch(e){if(e.code!=='EEXIST')throw e;const pid=Number(readFileSync(lockPath,'utf8'));try{process.kill(pid,0);throw Error('Another bounded cinema worker is active')}catch(e){if(e.code!=='ESRCH')throw e}unlinkSync(lockPath);lock=openSync(lockPath,'wx');writeFileSync(lock,String(process.pid))}
let child,timer,stopReason=null,peak=0,peakSwapGrowth=0,swapWarning=false;const log=openSync(root+'/logs/'+label+'.log','a'),start=Date.now();
const swap=()=>{const m=execFileSync('/usr/sbin/sysctl',['vm.swapusage'],{encoding:'utf8'}).match(/used = ([\d.]+)([MG])/);return m?Number(m[1])*(m[2]==='G'?1024:1):0};
const stop=reason=>{if(stopReason)return;stopReason=reason;try{process.kill(-child.pid,'SIGTERM')}catch{}};
try{const initial=swap();execFileSync('/bin/ps',['-axo','pid=,ppid=,rss=']);execFileSync('/usr/bin/memory_pressure',['-Q']);

process.once('SIGINT',()=>stop('Interrupted'));process.once('SIGTERM',()=>stop('Terminated'));
timer=setInterval(()=>{try{const rows=execFileSync('/bin/ps',['-axo','pid=,ppid=,rss='],{encoding:'utf8'}).trim().split('\n').map(s=>s.trim().split(/\s+/).map(Number)),owned=new Set([child.pid]);for(let i=0;i<8;i++)for(const[pid,ppid]of rows)if(owned.has(ppid))owned.add(pid);peak=Math.max(peak,rows.reduce((n,[pid,,rss])=>n+(owned.has(pid)?rss:0),0)/1024);if(peak>2560)stop('Worker tree exceeded 2.5GiB');const m=execFileSync('/usr/bin/memory_pressure',['-Q'],{encoding:'utf8'}).match(/free percentage: (\d+)/);const free=m?Number(m[1]):0;if(!m||free<20)stop('System memory free below 20percent');peakSwapGrowth=Math.max(peakSwapGrowth,swap()-initial);if(peakSwapGrowth>512){swapWarning=true;if(free<30)stop('Shared swap growth with free memory below 30percent')}}catch(e){stop('Cannot enforce memory guard: '+e.message)}},1000);

}finally{clearInterval(timer);closeSync(log);closeSync(lock);if(existsSync(lockPath)&&Number(readFileSync(lockPath,'utf8'))===process.pid)unlinkSync(lockPath)}
